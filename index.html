# ══════════════════════════════════════════════ # Queue Simulation — Single Server # ══════════════════════════════════════════════ # ── Input Data ──────────────────────────────── interarrival_times = [0.4, 1.2, 0.5, 1.7, 0.2, 1.2, 0.4, 1.6] service_times = [2.0, 0.7, 0.2, 1.1, 3.7] n = len(interarrival_times) # ── Initialize Lists ────────────────────────── arrival_time = [] service_start = [] service_end = [] waiting_time = [] server_idle = [] time_in_system = [] # ── Simulation Loop ─────────────────────────── cumulative_time = 0 for i in range(n): # Step 1: Arrival Time cumulative_time += interarrival_times[i] arrival_time.append(round(cumulative_time, 4)) if i < len(service_times): # Step 2: Service Start if i == 0: start = arrival_time[i] idle = arrival_time[i] else: start = max(arrival_time[i], service_end[i-1]) idle = max(0, arrival_time[i] - service_end[i-1]) # Step 3: Service End end = round(start + service_times[i], 4) # Step 4: Waiting Time wait = round(start - arrival_time[i], 4) # Step 5: Time in System tis = round(end - arrival_time[i], 4) service_start.append(round(start, 4)) service_end.append(end) waiting_time.append(wait) server_idle.append(round(idle, 4)) time_in_system.append(tis) else: service_start.append(None) service_end.append(None) waiting_time.append(None) server_idle.append(None) time_in_system.append(None) # ── Print Results ───────────────────────────── print("=" * 75) print("QUEUE SIMULATION RESULTS") print("=" * 75) print(f"{'#':<4} {'Arrival':<10} {'Start':<10} {'End':<10} {'Wait':<10} {'Idle':<10} {'TIS'}") print("-" * 75) for i in range(n): has = service_start[i] is not None print(f"{i+1:<4} {arrival_time[i]:<10} " f"{str(service_start[i]) if has else '—':<10} " f"{str(service_end[i]) if has else '—':<10} " f"{str(waiting_time[i]) if has else '—':<10} " f"{str(server_idle[i]) if has else '—':<10} " f"{str(time_in_system[i]) if has else '—'}") # ── Summary Statistics ──────────────────────── print("=" * 75) served_wait = [w for w in waiting_time if w is not None] served_svc = [s for s in service_times] served_tis = [t for t in time_in_system if t is not None] served_idle = [d for d in server_idle if d is not None] sim_end = service_end[len(service_times) - 1] utilization = (sim_end - sum(served_idle)) / sim_end * 100 print(f"Avg Waiting Time : {sum(served_wait)/len(served_wait):.4f} min") print(f"Avg Service Time : {sum(served_svc)/len(served_svc):.4f} min") print(f"Avg Time in System : {sum(served_tis)/len(served_tis):.4f} min") print(f"Total Idle Time : {sum(served_idle):.4f} min") print(f"Simulation End : {sim_end} min") print(f"Server Utilization : {utilization:.2f}%") print("=" * 75)
